🔍 TABLAS ENCONTRADAS:
==================================================

📋 Tabla: grados
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - nombre: VARCHAR

📋 Tabla: asignaturas
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - nombre: VARCHAR
    - creditos: INTEGER
    - horas_semanales: INTEGER
    - curso: INTEGER
    - cuatrimestre: INTEGER

📋 Tabla: profesores
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - nombre: VARCHAR
    - disponibilidad: TEXT

📋 Tabla: aulas
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - nombre: VARCHAR
    - capacidad: INTEGER
    - tipo: VARCHAR

📋 Tabla: menciones
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - nombre: VARCHAR
    - grado_id: INTEGER

📋 Tabla: asignatura_grado
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - asignatura_id: INTEGER
    - grado_id: INTEGER

📋 Tabla: profesor_asignatura
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - profesor_id: INTEGER
    - asignatura_id: INTEGER

📋 Tabla: sesiones
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - asignatura_id: INTEGER
    - profesor_id: INTEGER
    - aula_id: INTEGER
    - dia: VARCHAR
    - hora_inicio: TIME
    - hora_fin: TIME

📋 Tabla: restricciones
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - tipo: VARCHAR
    - valor: TEXT
    - asignatura_id: INTEGER
    - profesor_id: INTEGER
    - aula_id: INTEGER

📋 Tabla: asignatura_mencion
  Columnas:
    - id: INTEGER (PK) NOT NULL
    - asignatura_id: INTEGER
    - mencion_id: INTEGER

🔗 CLAVES FORÁNEAS:
==================================================

📋 menciones:
    grado_id -> grados.id

📋 asignatura_grado:
    grado_id -> grados.id
    asignatura_id -> asignaturas.id

📋 profesor_asignatura:
    asignatura_id -> asignaturas.id
    profesor_id -> profesores.id

📋 sesiones:
    aula_id -> aulas.id
    profesor_id -> profesores.id
    asignatura_id -> asignaturas.id

📋 restricciones:
    aula_id -> aulas.id
    profesor_id -> profesores.id
    asignatura_id -> asignaturas.id

📋 asignatura_mencion:
    mencion_id -> menciones.id
    asignatura_id -> asignaturas.id
